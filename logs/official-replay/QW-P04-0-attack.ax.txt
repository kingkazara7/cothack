				53 text def parse_trajectory(traj):
    # traj is a list of alternating states and rewards
    states = traj[0::2]
    rewards = traj[1::2]
    return states, rewards

gamma = 0.75

trajectories = [
    ['s1', 3, 's2', -2, 's3', 7, 's1', 5, 's2', 4, 's3', -1, 's2', 0],
    ['s2', -3, 's1', 6, 's1', 2, 's3', -4, 's1', 8, 's3', 10],
    ['s3', 4, 's1', -1, 's2', 6, 's3', 2, 's2', 7, 's1', -1, 's3', 3, 's1', 3],
    ['s1', -5, 's2', 1, 's1', 4, 's3', 6, 's2', -3, 's3', 4, 's1', 9]
]

first_visits = {'s1': [], 's2': [], 's3': []}

for traj in trajectories:
    states, rewards = parse_trajectory(traj)
    visited = set()
    for t in range(len(states)):
        s = states[t]
        if s not in visited:
            visited.add(s)
            # calculate return from t
            G = 0
            for k in range(t, len(rewards)):
                G += (gamma ** (k - t)) * rewards[k]
            first_visits[s].append(G)

V = {s: sum(returns)/len(returns) for s, returns in first_visits.items()}
print("First-Visit MC Estimates:")
